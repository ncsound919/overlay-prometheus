    14→// ─── Small random helpers ────────────────────────────────────────────────────
    15→
    16→export function rand(min: number, max: number): number {
    17→  return min + Math.random() * (max - min)
    18→}
    19→
    20→export function randInt(min: number, max: number): number {
    21→  return Math.floor(rand(min, max + 1))
    22→}
    23→
    24→export function pick<T>(arr: readonly T[]): T {
    25→  return arr[Math.floor(Math.random() * arr.length)]
    26→}
    27→
    28→export function chance(p: number): boolean {
    29→  return Math.random() < p
    30→}
    31→
    32→export function clamp01(v: number): number {
    33→  if (Number.isNaN(v)) return 0
    34→  return Math.min(1, Math.max(0, v))
    35→}
    36→
    37→export function round2(v: number): number {
    38→  return Math.round(v * 100) / 100
    39→}
    40→
    41→export function round3(v: number): number {
    42→  return Math.round(v * 1000) / 1000
    43→}

    66→// ─── Signal generation (Layer 1 + feature scoring Layer 2) ───────────────────
    67→
    68→export interface GeneratedSignal {
    69→  title: string
    70→  content: string
    71→  features: SignalFeatures
    72→  anomalyScore: number
    73→  noveltyScore: number
    74→  promoted: boolean
    75→}
    76→
    77→// Heavy-tail anomaly distribution: ~15% of signals exceed 0.75.
    78→function drawAnomalyScore(): number {
    79→  return chance(0.85) ? clamp01(Math.random() * 0.72) : clamp01(0.75 + Math.random() * 0.25)
    80→}
    81→
    82→function drawFeatures(anomaly: number): SignalFeatures {
    83→  return {
    84→    // entropy/gradient correlate positively with anomaly
    85→    entropy: round3(clamp01(0.25 + anomaly * 0.55 + rand(-0.12, 0.12))),
    86→    periodicity: round3(clamp01(rand(0.05, 0.6) - anomaly * 0.25)),
    87→    gradient: round3(clamp01(0.1 + anomaly * 0.7 + rand(-0.1, 0.1))),
    88→    dimensionality: randInt(2, 17),
    89→  }
    90→}
    91→
    92→function drawNoveltyScore(anomaly: number): number {
    93→  return round3(clamp01(0.15 + Math.random() * 0.55 + anomaly * 0.3))
    94→}

   491→function buildCounterfactuals(a: string, b: string): Counterfactual[] {
   492→  const cfs: Counterfactual[] = [
   493→    {
   494→      question: `Does the effect persist when ${a} variance is ablated?`,
   495→      result: `Simulated ablation reduces coupling strength by ${rand(30, 80).toFixed(0)}% (n=${randInt(200, 900)} runs).`,
   496→      supports: true,
   497→    },
   498→    {
   499→      question: `Is the correlation an artifact of sampling density?`,
   500→      result: `Bootstrap resampling at ${randInt(2, 9)}× density preserves the link (p=${rand(0.002, 0.04).toFixed(3)}).`,
   501→      supports: true,
   502→    },
   503→  ]
   504→  if (chance(0.35)) {
   505→    cfs.push({
   506→      question: `Would a shuffled null model of ${b} streams reproduce the pattern?`,
   507→      result: `Shuffled nulls reproduce the pattern in ${rand(6, 22).toFixed(0)}% of seeds — coupling likely spurious.`,
   508→      supports: false,
   509→    })
   510→  } else {
   511→    cfs.push({
   512→      question: `Would a shuffled null model of ${b} streams reproduce the pattern?`,
   513→      result: `Null model fails to reproduce the pattern in ${randInt(180, 480)}/${randInt(480, 500)} seeds.`,
   514→      supports: true,
   515→    })
   516→  }
   517→  return cfs
   518→}

   520→function twoDistinctDomains(available: string[]): [Category, Category] {
   521→  const pool = available.filter(isCategory)
   522→  const a = pool.length > 0 ? pick(pool) : pick(CATEGORIES)
   523→  let b = pick(CATEGORIES)
   524→  let guard = 0
   525→  while (b === a && guard++ < 10) b = pick(CATEGORIES)
   526→  return [a, b]
   527→}

   633→    if (!components.includes(c)) components.push(c)

   635→  return {
   636→    components,
   637→    mechanism: `${cap(phenom[0] ?? 'cross-domain coupling')} is stabilized by ${mech}, propagating through ${phenom.slice(1).join(' and ') || 'adjacent domains'} into a deployable ${solutionType} intervention.`,

   674→export function buildGovernanceContent(solutionType: SolutionType, domains: string[]): GovernanceContent {
   675→  const [lo, hi] = BASE_RISK[solutionType]
   676→  const riskScore = round3(clamp01(rand(lo, hi) + rand(-0.05, 0.05)))
   677→
   678→  const flags = new Set<string>()
   679→  for (const f of FLAGS_BY_TYPE[solutionType]) if (chance(0.75)) flags.add(f)
   680→  if (domains.includes('clinical') && solutionType !== 'protocol') flags.add('patient-safety review')
   681→  if (domains.includes('markets') && solutionType !== 'financial') flags.add('market-manipulation risk')
   682→  if (chance(0.3)) flags.add(pick(GENERIC_FLAGS))
   683→  if (flags.size === 0) flags.add(pick(FLAGS_BY_TYPE[solutionType]))
   684→
   685→  const flagList = [...flags]
   686→  const notes =
   687→    riskScore < 0.35
   688→      ? `Automated review: risk posture acceptable (${solutionType} class). ${flagList.length ? `Monitored flags: ${flagList.join(', ')}.` : 'No flags raised.'}`
   689→      : `Automated review escalated: ${solutionType} class exceeds the risk budget. ${flagList.join(', ')} require human sign-off before deployment.`
   690→  return { riskScore, ethicalFlags: flagList, notes }
   691→}
