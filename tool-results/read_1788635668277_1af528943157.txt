   156→   156→export function parseConfig(raw: string | null | undefined): EngineConfig {
   157→   157→  const config: EngineConfig = { tickIntervalMs: 6000, llmDiscovery: true, aggression: 0.7 }
   158→   158→  if (!raw) return config
   159→   159→  try {
   160→   160→    const parsed: unknown = JSON.parse(raw)
   161→   161→    if (typeof parsed !== 'object' || parsed === null) return config
   162→   162→    const rec = parsed as Record<string, unknown>
   163→   163→    if (typeof rec.tickIntervalMs === 'number' && Number.isFinite(rec.tickIntervalMs)) {
   164→   164→      config.tickIntervalMs = Math.min(120_000, Math.max(2_000, Math.round(rec.tickIntervalMs)))
   165→   165→    }
   166→   166→    if (typeof rec.llmDiscovery === 'boolean') config.llmDiscovery = rec.llmDiscovery
   167→   167→    if (typeof rec.aggression === 'number' && Number.isFinite(rec.aggression)) {
   168→   168→      config.aggression = Math.min(1, Math.max(0, rec.aggression))
   169→   169→    }
   170→   170→    return config
   171→   171→  } catch {
   172→   172→    return config
   173→   173→  }
   174→   174→}

   344→   344→async function runInputLayers(cycle: number): Promise<IngestSummary> {
   345→   345→  const streams = await db.domainStream.findMany({ where: { status: 'active' } })
   346→   346→  const signals: Array<{
   347→   347→    streamId: string
   348→   348→    category: string
   349→   349→    title: string
   350→   350→    content: string
   351→   351→    features: string
   352→   352→    anomalyScore: number
   353→   353→    noveltyScore: number
   354→   354→    cycle: number
   355→   355→    promoted: boolean
   356→   356→  }> = []
   357→   357→  const perStreamCounts: Array<{ id: string; count: number; velocity: number; health: number }> = []
   358→   358→
   359→   359→  for (const stream of streams) {
   360→   360→    const count = randInt(1, 4)
   361→   361→    const generated = generateSignals(stream.category, count)
   362→   362→    for (const g of generated) {
   363→   363→      signals.push({
   364→   364→        streamId: stream.id,
   365→   365→        category: stream.category,
   366→   366→        title: g.title,
   367→   367→        content: g.content,
   368→   368→        features: JSON.stringify(g.features),
   369→   369→        anomalyScore: g.anomalyScore,
   370→   370→        noveltyScore: g.noveltyScore,
   371→   371→        cycle,
   372→   372→        promoted: false,
   373→   373→      })
   374→   374→    }
   375→   375→    // Layer 1 stream bookkeeping: velocity is an EWMA of batch pressure.
   376→   376→    perStreamCounts.push({
   377→   377→      id: stream.id,
   378→   378→      count,
   379→   379→      velocity: round3(clamp01(stream.velocity * 0.6 + (count / 4) * 0.4)),
   380→   380→      health: round3(clamp01(Math.max(0.55, stream.health + rand(-0.03, 0.03)))),
   381→   381→    })
   382→   382→  }
   383→   383→
   384→   384→  // Layer 2: promote the top anomalies (cap 3/cycle; always promote at least one).
   385→   385→  const order = signals
   386→   386→    .map((s, i) => ({ i, score: s.anomalyScore }))
   387→   387→    .sort((a, b) => b.score - a.score)
   388→   388→  let promotedCount = 0
   389→   389→  for (const { i, score } of order) {
   390→   390→    if (promotedCount >= 3 || score < 0.8) break
   391→   391→    signals[i].promoted = true
   392→   392→    promotedCount++
   393→   393→  }
   394→   394→  if (promotedCount === 0 && order.length > 0) {
   395→   395→    signals[order[0].i].promoted = true
   396→   396→    promotedCount = 1
   397→   397→  }
   398→   398→
   399→   399→  if (signals.length > 0) await db.rawSignal.createMany({ data: signals })
   400→   400→  for (const s of perStreamCounts) {
   401→   401→    await db.domainStream.update({
   402→   402→      where: { id: s.id },
   403→   403→      data: {
   404→   404→        itemsIngested: { increment: s.count },
   405→   405→        velocity: s.velocity,
   406→   406→        health: s.health,
   407→   407→      },
   408→   408→    })
   409→   409→  }
   410→   410→
   411→   411→  const anomalyScores = signals.map((s) => s.anomalyScore)
   412→   412→  const noveltyScores = signals.map((s) => s.noveltyScore)
   413→   413→  const avg = (arr: number[]) => (arr.length ? arr.reduce((a, b) => a + b, 0) / arr.length : 0)
   414→   414→
   415→   415→  return {
   416→   416→    ingested: signals.length,
   417→   417→    streams: streams.length,
   418→   418→    anomalies: anomalyScores.filter((v) => v >= ANOMALY_THRESHOLD).length,
   419→   419→    promoted: promotedCount,
   420→   420→    avgAnomaly: round3(avg(anomalyScores)),
   421→   421→    avgNovelty: round3(avg(noveltyScores)),
   422→   422→  }
   423→   423→}

   443→   443→
   444→   444→function simulateValidation(
   445→   445→  h: CandidateHypothesis,
   446→   446→  aggression: number,
   447→   447→): { validation: ValidationStats; status: 'validated' | 'rejected' } {
   448→   448→  const simulations = randInt(200, 2000)
   449→   449→  const threshold = Math.min(0.6, 0.05 + h.confidence * aggression)
   450→   450→  const successBranch = Math.random() < clamp01(0.2 + h.confidence * 0.4 * aggression)
   451→   451→  const pValue = successBranch
   452→   452→    ? Math.pow(Math.random(), 3) * (0.05 + h.confidence * aggression)
   453→   453→    : rand(0.05, 0.9)
   454→   454→  const validated = pValue < threshold
   455→   455→  const passRate = clamp01(1 - pValue * 1.6 + rand(-0.04, 0.08))
   456→   456→  const passed = Math.min(simulations, Math.max(0, Math.round(simulations * passRate)))
   457→   457→  return {
   458→   458→    validation: {
   459→   459→      simulations,
   460→   460→      passed,
   461→   461→      failed: simulations - passed,
   462→   462→      pValue: round3(Math.min(1, Math.max(0.001, pValue))),
   463→   463→      effectSize: validated ? round3(rand(0.4, 1.6)) : round3(rand(0.02, 0.3)),
   464→   464→    },
   465→   465→    status: validated ? 'validated' : 'rejected',
   466→   466→  }
   467→   467→}

   469→   469→async function runDiscoveryLayer(
   470→   470→  cycle: number,
   471→   471→  aggression: number,
   472→   472→  llmDiscovery: boolean,
   473→   473→): Promise<{ hypotheses: DiscoveredHypothesis[]; llmUsed: boolean }> {
   474→   474→  const hot = await db.rawSignal.findMany({
   475→   475→    where: {
   476→   476→      cycle: { gte: cycle - 1 },
   477→   477→      OR: [{ promoted: true }, { anomalyScore: { gte: ANOMALY_THRESHOLD } }],
   478→   478→    },
   479→   479→    orderBy: { anomalyScore: 'desc' },
   480→   480→    take: 8,
   481→   481→  })
   482→   482→
   483→   483→  let candidates: CandidateHypothesis[] = []
   484→   484→  let llmUsed = false
   485→   485→
   486→   486→  if (llmDiscovery && hot.length > 0) {
   487→   487→    const llm = await generateHypothesesWithLlm(hot.map((s) => s.title))
   488→   488→    if (llm && llm.length > 0) {
   489→   489→      candidates = llm
   490→   490→      llmUsed = true
   491→   491→    }
   492→   492→  }
   493→   493→  if (candidates.length === 0) {
   494→   494→    const categories = hot.map((s) => s.category)
   495→   495→    candidates = generateTemplateHypotheses(categories, randInt(2, 3))
   496→   496→  }
   497→   497→
   498→   498→  const sourceSignals = hot.slice(0, 4).map((s) => s.title)
   499→   499→
   500→   500→  const created = await Promise.all(
   501→   501→    candidates.map(async (c) => {
   502→   502→      const row = await db.hypothesis.create({
   503→   503→        data: {
   504→   504→          title: c.title.slice(0, 300),
   505→   505→          summary: c.summary.slice(0, 2000),
   506→   506→          domains: JSON.stringify(c.domains),
   507→   507→          novelty: c.novelty,
   508→   508→          impact: c.impact,
   509→   509→          confidence: c.confidence,
   510→   510→          status: 'testing',
   511→   511→          counterfactuals: JSON.stringify(c.counterfactuals),
   512→   512→          sourceSignals: JSON.stringify(sourceSignals),
   513→   513→          generator: llmUsed ? 'llm' : 'template',
   514→   514→          cycle,
   515→   515→        },
   516→   516→      })
   517→   517→      return row
   518→   518→    }),
   519→   519→  )
   520→   520→
   521→   521→  const hypotheses: DiscoveredHypothesis[] = []
   522→   522→  for (let i = 0; i < created.length; i++) {
   523→   523→    const { validation, status } = simulateValidation(candidates[i], aggression)
   524→   524→    await db.hypothesis.update({ where: { id: created[i].id }, data: { status, validation: JSON.stringify(validation) } })
   525→   525→    hypotheses.push({ id: created[i].id, data: candidates[i], validation, status })
   526→   526→  }
   527→   527→
   528→   528→  return { hypotheses, llmUsed }
   529→   529→}

   575→   575→// ─── runTick: the full 8-layer pipeline ──────────────────────────────────────
   576→   576→
   577→   577→async function runTick(): Promise<TickResult> {
   578→   578→  const startedAt = Date.now()
   579→   579→  const state = await ensureEngine()
   580→   580→  const config = parseConfig(state.config)
   581→   581→  const cycle = state.cycles + 1
   582→   582→
   583→   583→  // ── Layer 1 + 2: INPUT UNIVERSE + PREPROCESSING ──
   584→   584→  const ingest = await runInputLayers(cycle)
   585→   585→  const metricWrites: Array<{ layer: number; key: string; value: number }> = [
   586→   586→    { layer: 1, key: 'ingested', value: ingest.ingested },
   587→   587→    { layer: 1, key: 'streams', value: ingest.streams },
   588→   588→    { layer: 1, key: 'anomalies', value: ingest.anomalies },
   589→   589→    { layer: 2, key: 'promoted', value: ingest.promoted },
   590→   590→    { layer: 2, key: 'avgAnomaly', value: ingest.avgAnomaly },
   591→   591→    { layer: 2, key: 'avgNovelty', value: ingest.avgNovelty },
   592→   592→  ]
   593→   593→
   594→   594→  // ── Layer 3: MULTI-DOMAIN ANALYSIS ──
   595→   595→  const streamRows = await db.domainStream.findMany({ where: { status: 'active' }, select: { category: true } })
   596→   596→  const domainPool = streamRows.map((s) => s.category).filter(isCategory)
   597→   597→  const linkCount = randInt(2, 4)
   598→   598→  for (let i = 0; i < linkCount; i++) {
   599→   599→    let source = domainPool.length >= 2 ? pick(domainPool) : pick(CATEGORIES)
   600→   600→    let target = domainPool.length >= 2 ? pick(domainPool) : pick(CATEGORIES)
   601→   601→    let guard = 0
   602→   602→    while (target === source && guard++ < 10) target = domainPool.length >= 2 ? pick(domainPool) : pick(CATEGORIES)
   603→   603→    if (target === source) continue
   604→   604→    await db.crossDomainLink.create({
   605→   605→      data: {
   606→   606→        sourceDomain: source,
   607→   607→        targetDomain: target,
   608→   608→        strength: round3(rand(0.4, 0.98)),
   609→   609→        mechanism: mechanismBetween(source, target),
   610→   610→        cycle,
   611→   611→      },
   612→   612→    })
   613→   613→  }
   614→   614→  const velocity = round3(clamp01(rand(0.15, 0.85) + ingest.avgAnomaly * 0.2))
   615→   615→  const acceleration = round3(rand(-0.2, 0.35))
   616→   616→  metricWrites.push(
   617→   617→    { layer: 3, key: 'crossLinks', value: linkCount },
   618→   618→    { layer: 3, key: 'velocity', value: velocity },
   619→   619→    { layer: 3, key: 'acceleration', value: acceleration },
   620→   620→  )
   621→   621→
   622→   622→  // ── Layer 4: DISCOVERY (LLM-enhanced, template fallback) ──
   623→   623→  const { hypotheses, llmUsed } = await runDiscoveryLayer(cycle, config.aggression, config.llmDiscovery)
   624→   624→  const validated = hypotheses.filter((h) => h.status === 'validated')
   625→   625→  metricWrites.push(
   626→   626→    { layer: 4, key: 'generated', value: hypotheses.length },
   627→   627→    { layer: 4, key: 'validated', value: validated.length },
   628→   628→  )
   629→   629→
   630→   630→  // ── Layer 5: BREAKTHROUGH SYNTHESIS ──
   631→   631→  const newBreakthroughs: Array<{
   632→   632→    id: string
   633→   633→    title: string
   634→   634→    solutionType: string
   635→   635→    impactRank: number
   636→   636→    gov: ReturnType<typeof buildGovernanceContent>
   637→   637→  }> = []
   638→   638→  for (const h of validated) {
   639→   639→    if (h.data.impact < 0.6) continue
   640→   640→    const solutionType = deriveSolutionType(h.data.domains)
   641→   641→    const synthesis = buildSynthesis(solutionType, h.data.domains)
   642→   642→    const impactRank = round3(h.data.impact * 0.7 + h.data.novelty * 0.3)
   643→   643→    const gov = buildGovernanceContent(solutionType, h.data.domains)
   644→   644→    const breakthrough = await db.breakthrough.create({
   645→   645→      data: {
   646→   646→        hypothesisId: h.id,
   647→   647→        title: h.data.title.slice(0, 300),
   648→   648→        solutionType,
   649→   649→        summary: buildBreakthroughSummary(solutionType, synthesis),
   650→   650→        synthesis: JSON.stringify(synthesis),
   651→   651→        impactRank,
   652→   652→        riskScore: gov.riskScore,
   653→   653→        status: 'synthesized',
   654→   654→        cycle,
   655→   655→      },
   656→   656→    })
   657→   657→    newBreakthroughs.push({
   658→   658→      id: breakthrough.id,
   659→   659→      title: breakthrough.title,
   660→   660→      solutionType,
   661→   661→      impactRank,
   662→   662→      gov,
   663→   663→    })
   664→   664→    await db.hypothesis.update({ where: { id: h.id }, data: { status: 'promoted' } })
   665→   665→  }
   666→   666→  metricWrites.push({ layer: 5, key: 'synthesized', value: newBreakthroughs.length })
   667→   667→
   668→   668→  // ── Layer 6: AUTONOMOUS OPTIMIZATION ──
   669→   669→  const successRate = hypotheses.length > 0 ? round3(validated.length / hypotheses.length) : 0
   670→   670→  const blindSpotScore = round3(clamp01(rand(0.05, 0.6)))
   671→   671→  metricWrites.push(
   672→   672→    { layer: 6, key: 'successRate', value: successRate },
   673→   673→    { layer: 6, key: 'avgNovelty', value: ingest.avgNovelty },
   674→   674→    { layer: 6, key: 'blindSpotScore', value: blindSpotScore },
   675→   675→  )
   676→   676→
   677→   677→  // ── Layer 7: GOVERNANCE ──
   678→   678→  let governed = 0
   679→   679→  let flagged = 0
   680→   680→  for (const bt of newBreakthroughs) {
   681→   681→    const passed = bt.gov.riskScore < 0.35
   682→   682→    await db.governanceCheck.create({
   683→   683→      data: {
   684→   684→        targetType: 'breakthrough',
   685→   685→        targetId: bt.id,
   686→   686→        targetTitle: bt.title,
   687→   687→        riskScore: bt.gov.riskScore,
   688→   688→        ethicalFlags: JSON.stringify(bt.gov.ethicalFlags),
   689→   689→        validationStatus: passed ? 'passed' : 'flagged',
   690→   690→        humanReview: !passed,
   691→   691→        notes: bt.gov.notes,
   692→   692→        cycle,
   693→   693→      },
   694→   694→    })
   695→   695→    if (passed) {
   696→   696→      governed++
   697→   697→      await db.breakthrough.update({ where: { id: bt.id }, data: { status: 'governed' } })
   698→   698→    } else {
   699→   699→      flagged++
   700→   700→    }
   701→   701→  }
   702→   702→  metricWrites.push(
   703→   703→    { layer: 7, key: 'checked', value: newBreakthroughs.length },
   704→   704→    { layer: 7, key: 'passed', value: governed },
   705→   705→    { layer: 7, key: 'flagged', value: flagged },
   706→   706→  )
   707→   707→
   708→   708→  // ── Layer 8: INSIGHT OUTPUT ──
   709→   709→  // cycle-summary every cycle + alerts (impactRank > 0.8) + blind-spot deep-dive (10%)
   710→   710→  const reports: Array<{ title: string; reportType: string; content: string }> = [
   711→   711→    {
   712→   712→      title: `Cycle ${cycle} Summary`,
   713→   713→      reportType: 'cycle-summary',
   714→   714→      content: buildCycleSummary({
   715→   715→        cycle,
   716→   716→        signals: ingest.ingested,
   717→   717→        anomalies: ingest.anomalies,
   718→   718→        hypotheses: hypotheses.length,
   719→   719→        validated: validated.length,
   720→   720→        breakthroughs: newBreakthroughs.length,
   721→   721→        governed,
   722→   722→        flagged,
   723→   723→      }),
   724→   724→    },
   725→   725→  ]
   726→   726→  for (const bt of newBreakthroughs) {
   727→   727→    if (bt.impactRank > 0.8) {
   728→   728→      const alert = buildAlertReport({
   729→   729→        title: bt.title,
   730→   730→        impactRank: bt.impactRank,
   731→   731→        solutionType: bt.solutionType,
   732→   732→        riskScore: bt.gov.riskScore,
   733→   733→        status: bt.gov.riskScore < 0.35 ? 'passed' : 'flagged',
   734→   734→      })
   735→   735→      reports.push({ ...alert, reportType: 'alert' })
   736→   736→    }
   737→   737→  }
   738→   738→  if (chance(0.1)) {
   739→   739→    const weakest = await db.domainStream.findFirst({ orderBy: { health: 'asc' } })
   740→   740→    if (weakest) {
   741→   741→      const deepDive = buildBlindSpotReport({
   742→   742→        cycle,
   743→   743→        streamName: weakest.name,
   744→   744→        health: weakest.health,
   745→   745→        velocity: weakest.velocity,
   746→   746→        successRate,
   747→   747→      })
   748→   748→      reports.push({ ...deepDive, reportType: 'deep-dive' })
   749→   749→    }
   750→   750→  }
   751→   751→
   752→   752→  const persistedReports = await Promise.all(
   753→   753→    reports.map((r) =>
   754→   754→      db.insightReport.create({
   755→   755→        data: { title: r.title, reportType: r.reportType, content: r.content, cycle },
   756→   756→      }),
   757→   757→    ),
   758→   758→  )
   759→   759→  const alertCount = persistedReports.filter((r) => r.reportType === 'alert').length
   760→   760→  metricWrites.push(
   761→   761→    { layer: 8, key: 'reportsPublished', value: persistedReports.length },
   762→   762→    { layer: 8, key: 'alerts', value: alertCount },
   763→   763→  )
   764→   764→
   765→   765→  // Persist all layer metrics.
   766→   766→  await db.layerMetric.createMany({ data: metricWrites.map((m) => ({ ...m, cycle })) })
   767→   767→
   768→   768→  // Close the cycle.
   769→   769→  await db.engineState.update({
   770→   770→    where: { id: ENGINE_ID },
   771→   771→    data: { cycles: cycle, lastTickAt: new Date() },
   772→   772→  })
   773→   773→
   774→   774→  // Prune (best-effort) to keep the SQLite file small.
   775→   775→  await pruneAll()
   776→   776→
   777→   777→  const result: TickResult = {
   778→   778→    cycle,
   779→   779→    signals: ingest.ingested,
   780→   780→    anomalies: ingest.anomalies,
   781→   781→    promoted: ingest.promoted,
   782→   782→    crossLinks: linkCount,
   783→   783→    hypotheses: hypotheses.length,
   784→   784→    validated: validated.length,
   785→   785→    rejected: hypotheses.length - validated.length,
   786→   786→    breakthroughs: newBreakthroughs.length,
   787→   787→    governed,
   788→   788→    flagged,
   789→   789→    reports: persistedReports.length,
   790→   790→    llmUsed,
   791→   791→    durationMs: Date.now() - startedAt,
   792→   792→  }
   793→   793→  console.log(
   794→   794→    `[engine/core] cycle ${cycle} done in ${result.durationMs}ms — signals=${result.signals} ` +
   795→   795→      `hypotheses=${result.hypotheses} (validated=${result.validated}) breakthroughs=${result.breakthroughs} ` +
   796→   796→      `llm=${result.llmUsed ? 'on' : 'off'}`,
   797→   797→  )
   798→   798→  return result
   799→   799→}

   915→   915→  const cfg = parseConfig(initial.config)
   916→   916→  const due =
   917→   917→    initial.status === 'running' && Date.now() - initial.lastTickAt.getTime() >= cfg.tickIntervalMs
   918→   918→  if (due && !tickMutex.promise) {
