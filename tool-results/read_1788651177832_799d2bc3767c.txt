   100→// ─── Sparkline (pure SVG, non-scaling stroke, gradient area) ─────────────────
   101→
   102→function Sparkline({
   103→  series,
   104→  tone,
   105→  height = 72,
   106→  label,
   107→  sub,
   108→}: {
   109→  series: number[];
   110→  tone: MeterTone;
   111→  height?: number;
   112→  label: string;
   113→  sub?: string;
   114→}) {
   115→  const W = 300;
   116→  const H = height;
   117→  const color = TONE_STROKE[tone];
   118→  const gradId = `lw-spark-${tone}-${label.replace(/[^a-z0-9]/gi, '')}`;
   119→
   120→  if (series.length < 2) {
   121→    return (
   122→      <div className="flex h-[72px] items-center justify-center rounded-lg border border-dashed border-zinc-800 text-[11px] text-zinc-600">
   123→        history accumulating — need ≥2 cycles
   124→      </div>
   125→    );
   126→  }
   127→
   128→  const max = Math.max(...series, 0.0001);
   129→  const min = Math.min(...series);
   130→  const span = max - min || 1;
   131→  const step = W / (series.length - 1);
   132→  const pts = series.map((v, i) => [i * step, H - 6 - ((v - min) / span) * (H - 14)] as const);
   133→  const line = pts.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(' ');
   134→  const area = `0,${H} ${line} ${W},${H}`;
   135→  const last = pts[pts.length - 1];
   136→
   137→  return (
   138→    <div className="relative">
   139→      <svg
   140→        viewBox={`0 0 ${W} ${H}`}
   141→        preserveAspectRatio="none"
   142→        className="h-[72px] w-full"
   143→        role="img"
   144→        aria-label={`${label} — ${series.length} points, latest ${fmtNum(series[series.length - 1], series[series.length - 1] < 10 ? 2 : 0)}`}
   145→      >
   146→        <defs>
   147→          <linearGradient id={gradId} x1="0" y1="0" x2="0" y2="1">
   148→            <stop offset="0%" stopColor={color} stopOpacity="0.35" />
   149→            <stop offset="100%" stopColor={color} stopOpacity="0" />
   150→          </linearGradient>
   151→        </defs>
   152→        {/* min/max guide rails */}
   153→        <line x1="0" y1="6" x2={W} y2="6" stroke="#3f3f46" strokeDasharray="3 5" strokeWidth="1" vectorEffect="non-scaling-stroke" />
   154→        <line x1="0" y1={H - 6} x2={W} y2={H - 6} stroke="#3f3f46" strokeDasharray="3 5" strokeWidth="1" vectorEffect="non-scaling-stroke" />
   155→        <polygon points={area} fill={`url(#${gradId})`} />
   156→        <polyline
   157→          points={line}
   158→          fill="none"
   159→          stroke={color}
   160→          strokeWidth="1.75"
   161→          strokeLinejoin="round"
   162→          strokeLinecap="round"
   163→          vectorEffect="non-scaling-stroke"
   164→        />
   165→        {/* last-point marker (r fixed — preserveAspectRatio=none would distort circles) */}
   166→        <circle cx={last[0]} cy={last[1]} r="3" fill={color} />
   167→      </svg>
   168→      <div className="pointer-events-none absolute inset-x-0 -top-0.5 flex items-center justify-between font-mono text-[9px] tabular-nums text-zinc-600">
   169→        <span>{sub ?? `max ${fmtNum(max, max < 10 ? 2 : 0)}`}</span>
   170→        <span className={TONE_TEXT[tone]}>{fmtNum(series[series.length - 1], series[series.length - 1] < 10 ? 2 : 0)}</span>
   171→      </div>
   172→    </div>
   173→  );
   174→}

   178→function layerSeries(
   179→  layer: number,
   180→  data: EngineStatePayload,
   181→): { label: string; series: number[]; tone: MeterTone; sub?: string } | null {
   182→  const spc = data.analytics?.signalsPerCycle ?? [];
   183→  switch (layer) {
   184→    case 1:
   185→      return spc.length
   186→        ? { label: 'signals / cycle', series: spc.map((p) => p.signals), tone: 'teal' as const }
   187→        : null;
   188→    case 2:
   189→      return spc.length
   190→        ? { label: 'anomalies / cycle', series: spc.map((p) => p.anomalies), tone: 'amber' as const, sub: `≥ ${0.75} score` }
   191→        : null;
   192→    case 3: {
   193→      // Correlations carry no cycle — render the strength spectrum instead.
   194→      const bands = [0, 0, 0];
   195→      for (const c of data.correlations) {
   196→        if (c.strength >= 0.8) bands[2]++;
   197→        else if (c.strength >= 0.6) bands[1]++;
   198→        else bands[0]++;
   199→      }
   200→      return { label: 'link strength spectrum', series: bands, tone: 'violet' as const, sub: '0.4–0.6 / 0.6–0.8 / 0.8+' };
   201→    }
   202→    case 4: {
   203→      const byCycle = new Map<number, number>();
   204→      for (const h of data.hypotheses) byCycle.set(h.cycle, (byCycle.get(h.cycle) ?? 0) + 1);
   205→      const series = [...byCycle.entries()].sort((a, b) => a[0] - b[0]).map(([, n]) => n);
   206→      return series.length >= 2 ? { label: 'hypotheses / cycle', series, tone: 'violet' as const } : null;
   207→    }
   208→    case 5: {
   209→      const byCycle = new Map<number, number>();
   210→      for (const b of data.breakthroughs) byCycle.set(b.cycle, (byCycle.get(b.cycle) ?? 0) + 1);
   211→      const series = [...byCycle.entries()].sort((a, b) => a[0] - b[0]).map(([, n]) => n);
   212→      return series.length >= 2 ? { label: 'breakthroughs / cycle', series, tone: 'emerald' as const } : null;
   213→    }
   214→    case 6:
   215→      return (data.analytics?.optimizationTrend ?? []).length >= 2
   216→        ? {
   217→            label: 'success rate trend',
   218→            series: data.analytics.optimizationTrend.map((t) => Math.round(t.successRate * 100)),
   219→            tone: 'emerald' as const,
   220→            sub: '% validated',
   221→          }
   222→        : null;
   223→    case 7: {
   224→      const byCycle = new Map<number, number>();
   225→      for (const g of data.governance) byCycle.set(g.cycle, (byCycle.get(g.cycle) ?? 0) + 1);
   226→      const series = [...byCycle.entries()].sort((a, b) => a[0] - b[0]).map(([, n]) => n);
   227→      return series.length >= 2 ? { label: 'checks / cycle', series, tone: 'amber' as const } : null;
   228→    }
   229→    default: {
   230→      const byCycle = new Map<number, number>();
   231→      for (const r of data.reports) byCycle.set(r.cycle, (byCycle.get(r.cycle) ?? 0) + 1);
   232→      const series = [...byCycle.entries()].sort((a, b) => a[0] - b[0]).map(([, n]) => n);
   233→      return series.length >= 2 ? { label: 'reports / cycle', series, tone: 'rose' as const } : null;
   234→    }
   235→  }
   236→}

   394→function isAutoTriaged(g: GovernanceCheck): boolean {
   395→  return typeof g.notes === 'string' && g.notes.startsWith('AUTO-TRIAGE');
   396→}
   397→
   398→function decisionChip(g: GovernanceCheck): { label: string; cls: string } {
   399→  if (g.reviewDecision === 'approved') {
   400→    return isAutoTriaged(g)
   401→      ? { label: 'auto ✓', cls: 'border-amber-500/40 bg-amber-500/10 text-amber-300' }
   402→      : { label: 'approved ✓', cls: 'border-emerald-500/40 bg-emerald-500/10 text-emerald-300' };
   403→  }
   404→  if (g.reviewDecision === 'rejected') {
   405→    return { label: 'rejected ✗', cls: 'border-rose-500/40 bg-rose-500/10 text-rose-300' };
   406→  }
   407→  return { label: 'pending…', cls: 'border-amber-500/40 bg-amber-500/10 text-amber-300' };
   408→}

   410→function GovernanceRows({ governance }: { governance: GovernanceCheck[] }) {
   411→  // Pendings pinned first (oldest breach at top), then decided rows newest-first.
   412→  const pend = governance.filter((g) => g.humanReview && !g.reviewDecision).slice().reverse();
   413→  const decided = governance.filter((g) => !g.humanReview || g.reviewDecision);
   414→  const ordered = [...pend, ...decided];
   415→  return (
   416→    <>
   417→      {ordered.map((g) => {
   418→        const d = decisionChip(g);
   419→        return (
   420→          <li key={g.id} className={ROW}>
   421→            <span className="min-w-0 flex-1 truncate text-xs text-zinc-200">{g.targetTitle}</span>
   422→            <Chip className={riskBadgeClass(g.riskScore)}>risk {g.riskScore.toFixed(2)}</Chip>
   423→            <Chip className={d.cls}>{d.label}</Chip>
   424→            {g.ethicalFlags.length > 0 ? (
   425→              <Chip className="border-rose-500/40 bg-rose-500/10 text-rose-300" title={g.ethicalFlags.join(', ')}>
   426→                ⚑ {g.ethicalFlags.length}
   427→              </Chip>
   428→            ) : null}
   429→            <CycleChip cycle={g.cycle} />
   430→          </li>
   431→        );
   432→      })}
   433→    </>
   434→  );
   435→}

   462→// ─── Citation web (L4): evidence signals → hypotheses ────────────────────────
   463→
   464→const CITE_ROW_H = 26;
   465→const CITE_W = 640;
   466→const CITE_MAX_SIGNALS = 12;
   467→const CITE_MAX_HYPS = 7;
   468→
   469→function clipTitle(t: string, max = 26): string {
   470→  return t.length > max ? `${t.slice(0, max)}…` : t;
   471→}
   472→
   473→/**
   474→ * Bipartite citation web for Layer 4: cited evidence signals (left, teal,
   475→ * ranked by how many of the rendered hypotheses cite them) vs the newest
   476→ * hypotheses (right, violet). Bezier edges = one citation each. Pure SVG,
   477→ * horizontally scrollable on narrow screens.
   478→ */
   479→function CitationWeb({ data }: { data: EngineStatePayload }) {
   480→  const hypos = data.hypotheses.slice(0, CITE_MAX_HYPS);
   481→  if (hypos.length === 0) return null;
   482→
   483→  const citeCount = new Map<string, number>();
   484→  for (const h of hypos) {
   485→    for (const t of h.sourceSignals ?? []) citeCount.set(t, (citeCount.get(t) ?? 0) + 1);
   486→  }
   487→  const ranked = [...citeCount.entries()].sort(
   488→    (a, b) => b[1] - a[1] || a[0].localeCompare(b[0]),
   489→  );
   490→  const shown = ranked.slice(0, CITE_MAX_SIGNALS);
   491→  const overflow = ranked.length - shown.length;
   492→  const signalIndex = new Map(shown.map(([t], i) => [t, i]));
   493→
   494→  const leftCy = (i: number) => 10 + i * CITE_ROW_H;
   495→  const rightCy = (i: number) => 10 + i * CITE_ROW_H;
   496→  const rows = Math.max(shown.length, hypos.length);
   497→  const H = rows * CITE_ROW_H + 14;
   498→
   499→  const edges: Array<{ key: string; d: string; tip: string }> = [];
   500→  hypos.forEach((h, hi) => {
   501→    for (const t of h.sourceSignals ?? []) {
   502→      const si = signalIndex.get(t);
   503→      if (si === undefined) continue;
   504→      const x1 = 306;
   505→      const y1 = leftCy(si);
   506→      const x2 = 334;
   507→      const y2 = rightCy(hi);
   508→      const mx = (x1 + x2) / 2;
   509→      edges.push({
   510→        key: `${h.id}::${si}::${t.slice(0, 16)}`,
   511→        d: `M ${x1} ${y1} C ${mx} ${y1}, ${mx} ${y2}, ${x2} ${y2}`,
   512→        tip: `${t} → ${h.title}`,
   513→      });
   514→    }
   515→  });
   516→
   517→  return (
   518→    <Panel
   519→      title="Citation Web · Evidence → Hypotheses"
   520→      icon={GitBranch}
   521→      ariaLabel="Citation web — evidence signals to hypotheses"
   522→      actions={
   523→        <span className="font-mono text-[10px] uppercase tracking-wider text-zinc-500">
   524→          {shown.length} signals · {hypos.length} hyps · {edges.length} edges
   525→        </span>
   526→      }
   527→    >
   528→      <div className="flex flex-wrap items-center gap-3 pb-2 font-mono text-[9px] uppercase tracking-wider text-zinc-600">
   529→        <span className="flex items-center gap-1.5">
   530→          <span className="size-2 rounded-sm border border-teal-500/40 bg-teal-500/10" aria-hidden /> cited signal
   531→        </span>
   532→        <span className="flex items-center gap-1.5">
   533→          <span className="size-2 rounded-sm border border-violet-500/40 bg-violet-500/10" aria-hidden /> hypothesis
   534→        </span>
   535→        <span className="flex items-center gap-1.5">
   536→          <svg width="18" height="6" aria-hidden>
   537→            <path d="M 0 3 C 6 3, 12 3, 18 3" stroke="#8b5cf6" strokeWidth="1.5" fill="none" opacity="0.6" />
   538→          </svg>
   539→          citation
   540→        </span>
   541→      </div>
   542→      <div className="mc-scroll overflow-x-auto">
   543→        <svg
   544→          viewBox={`0 0 ${CITE_W} ${H}`}
   545→          className="w-full min-w-[560px]"
   546→          role="img"
   547→          aria-label={`Citation web: ${shown.length} evidence signals linked to ${hypos.length} hypotheses via ${edges.length} citations`}
   548→        >
   549→          {/* edges */}
   550→          {edges.map((e) => (
   551→            <path
   552→              key={e.key}
   553→              d={e.d}
   554→              fill="none"
   555→              stroke="#8b5cf6"
   556→              strokeWidth="1.25"
   557→              opacity="0.3"
   558→              className="transition-opacity duration-200 hover:opacity-90"
   559→            >
   560→              <title>{e.tip}</title>
   561→            </path>
   562→          ))}
   563→          {/* left nodes — cited signals */}
   564→          {shown.map(([title], i) => {
   565→            const y = leftCy(i);
   566→            return (
   567→              <g key={`sig-${i}`} className="cursor-default">
   568→                <rect
   569→                  x="2"
   570→                  y={y - 9}
   571→                  width="304"
   572→                  height="18"
   573→                  rx="4"
   574→                  className="fill-zinc-950/80 stroke-teal-500/30 transition-colors duration-200 hover:stroke-teal-400/70"
   575→                >
   576→                  <title>{title}</title>
   577→                </rect>
   578→                <circle cx="10" cy={y} r="2" fill="#14b8a6" />
   579→                <text x="17" y={y + 3.5} fontSize="10" fontFamily="ui-monospace, monospace" fill="#a1a1aa">
   580→                  {clipTitle(title)}
   581→                </text>
   582→              </g>
   583→            );
   584→          })}
   585→          {/* right nodes — hypotheses */}
   586→          {hypos.map((h, i) => {
   587→            const y = rightCy(i);
   588→            return (
   589→              <g key={h.id} className="cursor-default">
   590→                <rect
   591→                  x="334"
   592→                  y={y - 9}
   593→                  width="304"
   594→                  height="18"
   595→                  rx="4"
   596→                  className="fill-zinc-950/80 stroke-violet-500/30 transition-colors duration-200 hover:stroke-violet-400/70"
   597→                >
   598→                  <title>{h.title}</title>
   599→                </rect>
   600→                <circle cx="342" cy={y} r="2" fill="#8b5cf6" />
   601→                <text x="349" y={y + 3.5} fontSize="10" fontFamily="ui-monospace, monospace" fill="#d4d4d8">
   602→                  {clipTitle(h.title)}
   603→                </text>
   604→              </g>
   605→            );
   606→          })}
   607→        </svg>
   608→      </div>
   609→      {overflow > 0 ? (
   610→        <p className="pt-1.5 font-mono text-[10px] text-zinc-600" aria-hidden>
   611→          +{overflow} lower-cited signal{overflow === 1 ? '' : 's'} not rendered
   612→        </p>
   613→      ) : null}
   614→    </Panel>
   615→  );
   616→}

   620→function layerEntities(layer: number, data: EngineStatePayload): number {
   621→  switch (layer) {
   622→    case 1: return data.streams.length;
   623→    case 2: return data.signals.length;
   624→    case 3: return data.correlations.length;
   625→    case 4: return data.hypotheses.length;
   626→    case 5: return data.breakthroughs.length;
   627→    case 6: return data.analytics?.optimizationTrend?.length ?? 0;
   628→    case 7: return data.governance.length;
   629→    default: return data.reports.length;
   630→  }
   631→}
