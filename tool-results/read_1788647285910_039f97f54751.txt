   144→/** Runs one tick; if a tick is already in flight, callers await that same promise. */
   145→export async function runTickSerialized(): Promise<TickResult> {
   146→  if (tickMutex.promise) return tickMutex.promise
   147→  const p = runTick().finally(() => {
   148→    tickMutex.promise = null
   149→  })
   150→  tickMutex.promise = p
   151→  return p
   152→}

   156→export function parseConfig(raw: string | null | undefined): EngineConfig {
   157→  const config: EngineConfig = { tickIntervalMs: 6000, llmDiscovery: true, aggression: 0.7 }
   158→  if (!raw) return config
   159→  try {
   160→    const parsed: unknown = JSON.parse(raw)
   161→    if (typeof parsed !== 'object' || parsed === null) return config
   162→    const rec = parsed as Record<string, unknown>
   163→    if (typeof rec.tickIntervalMs === 'number' && Number.isFinite(rec.tickIntervalMs)) {
   164→      config.tickIntervalMs = Math.min(120_000, Math.max(2_000, Math.round(rec.tickIntervalMs)))
   165→    }
   166→    if (typeof rec.llmDiscovery === 'boolean') config.llmDiscovery = rec.llmDiscovery
   167→    if (typeof rec.aggression === 'number' && Number.isFinite(rec.aggression)) {
   168→      config.aggression = Math.min(1, Math.max(0, rec.aggression))
   169→    }
   170→    return config
   171→  } catch {
   172→    return config
   173→  }
   174→}

   182→  if (typeof patch.aggression === 'number' && Number.isFinite(patch.aggression)) {

   235→function isRecord(v: unknown): v is Record<string, unknown> {
   236→  return typeof v === 'object' && v !== null && !Array.isArray(v)
   237→}

   239→function numOr(v: unknown, fallback: number): number {
   240→  return typeof v === 'number' && Number.isFinite(v) ? v : fallback
   241→}

   243→function parseNumberRecord(raw: string | null | undefined): Record<string, number> {
   244→  if (!raw) return {}
   245→  try {
   246→    const parsed: unknown = JSON.parse(raw)
   247→    if (!isRecord(parsed)) return {}
   248→    const out: Record<string, number> = {}
   249→    for (const [k, v] of Object.entries(parsed)) {
   250→      if (typeof v === 'number' && Number.isFinite(v)) out[k] = v
   251→    }
   252→    return out
   253→  } catch {
   254→    return {}
   255→  }
   256→}

   258→function parseStringArray(raw: string | null | undefined): string[] {
   259→  if (!raw) return []
   260→  try {
   261→    const parsed: unknown = JSON.parse(raw)
   262→    if (!Array.isArray(parsed)) return []
   263→    return parsed.filter((v): v is string => typeof v === 'string')
   264→  } catch {
   265→    return []
   266→  }
   267→}

   269→function parseCounterfactuals(raw: string | null | undefined): Counterfactual[] {
   270→  if (!raw) return []
   271→  try {
   272→    const parsed: unknown = JSON.parse(raw)
   273→    if (!Array.isArray(parsed)) return []
   274→    const out: Counterfactual[] = []
   275→    for (const item of parsed) {
   276→      if (!isRecord(item)) continue
   277→      if (typeof item.question !== 'string' || typeof item.result !== 'string') continue
   278→      out.push({
   279→        question: item.question,
   280→        result: item.result,
   281→        supports: item.supports === true,
   282→      })
   283→    }
   284→    return out
   285→  } catch {
   286→    return []
   287→  }
   288→}

   290→function parseValidation(raw: string | null | undefined): ValidationStats {
   291→  if (!raw) return { simulations: 0, passed: 0, failed: 0, pValue: 1, effectSize: 0 }
   292→  try {
   293→    const parsed: unknown = JSON.parse(raw)
   294→    if (!isRecord(parsed)) return { simulations: 0, passed: 0, failed: 0, pValue: 1, effectSize: 0 }
   295→    return {
   296→      simulations: Math.max(0, Math.round(numOr(parsed.simulations, 0))),
   297→      passed: Math.max(0, Math.round(numOr(parsed.passed, 0))),
   298→      failed: Math.max(0, Math.round(numOr(parsed.failed, 0))),
   299→      pValue: clamp01(numOr(parsed.pValue, 1)),
   300→      effectSize: numOr(parsed.effectSize, 0),
   301→    }
   302→  } catch {
   303→    return { simulations: 0, passed: 0, failed: 0, pValue: 1, effectSize: 0 }
   304→  }
   305→}

   344→async function runInputLayers(cycle: number): Promise<IngestSummary> {
   345→  const streams = await db.domainStream.findMany({ where: { status: 'active' } })
   346→  const signals: Array<{
   347→    streamId: string
   348→    category: string
   349→    title: string
   350→    content: string
   351→    features: string
   352→    anomalyScore: number
   353→    noveltyScore: number
   354→    cycle: number
   355→    promoted: boolean
   356→  }> = []
   357→  const perStreamCounts: Array<{ id: string; count: number; velocity: number; health: number }> = []
   358→
   359→  for (const stream of streams) {
   360→    const count = randInt(1, 4)
   361→    const generated = generateSignals(stream.category, count)
   362→    for (const g of generated) {
   363→      signals.push({
   364→        streamId: stream.id,
   365→        category: stream.category,
   366→        title: g.title,
   367→        content: g.content,
   368→        features: JSON.stringify(g.features),
   369→        anomalyScore: g.anomalyScore,
   370→        noveltyScore: g.noveltyScore,
   371→        cycle,
   372→        promoted: false,
   373→      })
   374→    }
   375→    // Layer 1 stream bookkeeping: velocity is an EWMA of batch pressure.
   376→    perStreamCounts.push({
   377→      id: stream.id,
   378→      count,
   379→      velocity: round3(clamp01(stream.velocity * 0.6 + (count / 4) * 0.4)),
   380→      health: round3(clamp01(Math.max(0.55, stream.health + rand(-0.03, 0.03)))),
   381→    })
   382→  }
   383→
   384→  // Layer 2: promote the top anomalies (cap 3/cycle; always promote at least one).
   385→  const order = signals
   386→    .map((s, i) => ({ i, score: s.anomalyScore }))
   387→    .sort((a, b) => b.score - a.score)
   388→  let promotedCount = 0
   389→  for (const { i, score } of order) {
   390→    if (promotedCount >= 3 || score < 0.8) break
   391→    signals[i].promoted = true
   392→    promotedCount++
   393→  }
   394→  if (promotedCount === 0 && order.length > 0) {
   395→    signals[order[0].i].promoted = true
   396→    promotedCount = 1
   397→  }
   398→
   399→  if (signals.length > 0) await db.rawSignal.createMany({ data: signals })
   400→  for (const s of perStreamCounts) {
   401→    await db.domainStream.update({
   402→      where: { id: s.id },
   403→      data: {
   404→        itemsIngested: { increment: s.count },
   405→        velocity: s.velocity,
   406→        health: s.health,
   407→      },
   408→    })
   409→  }
   410→
   411→  const anomalyScores = signals.map((s) => s.anomalyScore)
   412→  const noveltyScores = signals.map((s) => s.noveltyScore)
   413→  const avg = (arr: number[]) => (arr.length ? arr.reduce((a, b) => a + b, 0) / arr.length : 0)
   414→
   415→  return {
   416→    ingested: signals.length,
   417→    streams: streams.length,
   418→    anomalies: anomalyScores.filter((v) => v >= ANOMALY_THRESHOLD).length,
   419→    promoted: promotedCount,
   420→    avgAnomaly: round3(avg(anomalyScores)),
   421→    avgNovelty: round3(avg(noveltyScores)),
   422→  }
   423→}

   427→interface CandidateHypothesis {
   428→  title: string
   429→  summary: string
   430→  domains: string[]
   431→  novelty: number
   432→  impact: number
   433→  confidence: number

   437→interface DiscoveredHypothesis {
   438→  id: string
   439→  data: CandidateHypothesis
   440→  validation: ValidationStats
   441→  status: 'validated' | 'rejected'
   442→}
   443→
   444→function simulateValidation(
   445→  h: CandidateHypothesis,
   446→  aggression: number,
   447→): { validation: ValidationStats; status: 'validated' | 'rejected' } {
   448→  const simulations = randInt(200, 2000)
   449→  const threshold = Math.min(0.6, 0.05 + h.confidence * aggression)
   450→  const successBranch = Math.random() < clamp01(0.2 + h.confidence * 0.4 * aggression)
   451→  const pValue = successBranch
   452→    ? Math.pow(Math.random(), 3) * (0.05 + h.confidence * aggression)
   453→    : rand(0.05, 0.9)
   454→  const validated = pValue < threshold
   455→  const passRate = clamp01(1 - pValue * 1.6 + rand(-0.04, 0.08))
   456→  const passed = Math.min(simulations, Math.max(0, Math.round(simulations * passRate)))
   457→  return {
   458→    validation: {
   459→      simulations,
   460→      passed,
   461→      failed: simulations - passed,
   462→      pValue: round3(Math.min(1, Math.max(0.001, pValue))),
   463→      effectSize: validated ? round3(rand(0.4, 1.6)) : round3(rand(0.02, 0.3)),
   464→    },
   465→    status: validated ? 'validated' : 'rejected',
   466→  }
   467→}

   469→async function runDiscoveryLayer(
   470→  cycle: number,
   471→  aggression: number,
   472→  llmDiscovery: boolean,
   473→): Promise<{ hypotheses: DiscoveredHypothesis[]; llmUsed: boolean }> {
   474→  const hot = await db.rawSignal.findMany({
   475→    where: {
   476→      cycle: { gte: cycle - 1 },
   477→      OR: [{ promoted: true }, { anomalyScore: { gte: ANOMALY_THRESHOLD } }],
   478→    },
   479→    orderBy: { anomalyScore: 'desc' },
   480→    take: 8,
   481→  })
   482→
   483→  let candidates: CandidateHypothesis[] = []
   484→  let llmUsed = false
   485→
   486→  if (llmDiscovery && hot.length > 0) {
   487→    const llm = await generateHypothesesWithLlm(hot.map((s) => s.title))
   488→    if (llm && llm.length > 0) {
   489→      candidates = llm
   490→      llmUsed = true
   491→    }
   492→  }
   493→  if (candidates.length === 0) {
   494→    const categories = hot.map((s) => s.category)
   495→    candidates = generateTemplateHypotheses(categories, randInt(2, 3))
   496→  }
   497→
   498→  const sourceSignals = hot.slice(0, 4).map((s) => s.title)
   499→
   500→  const created = await Promise.all(
   501→    candidates.map(async (c) => {
   502→      const row = await db.hypothesis.create({
   503→        data: {
   504→          title: c.title.slice(0, 300),
   505→          summary: c.summary.slice(0, 2000),
   506→          domains: JSON.stringify(c.domains),
   507→          novelty: c.novelty,
   508→          impact: c.impact,
   509→          confidence: c.confidence,
   510→          status: 'testing',
   511→          counterfactuals: JSON.stringify(c.counterfactuals),
   512→          sourceSignals: JSON.stringify(sourceSignals),
   513→          generator: llmUsed ? 'llm' : 'template',
   514→          cycle,
   515→        },
   516→      })
   517→      return row
   518→    }),
   519→  )
   520→
   521→  const hypotheses: DiscoveredHypothesis[] = []
   522→  for (let i = 0; i < created.length; i++) {
   523→    const { validation, status } = simulateValidation(candidates[i], aggression)
   524→    await db.hypothesis.update({ where: { id: created[i].id }, data: { status, validation: JSON.stringify(validation) } })
   525→    hypotheses.push({ id: created[i].id, data: candidates[i], validation, status })
   526→  }
   527→
   528→  return { hypotheses, llmUsed }
   529→}

   533→async function pruneTable(
   534→  findRecent: (take: number, skip: number) => Promise<Array<{ createdAt: Date }>>,
   535→  deleteOlderThan: (createdAt: Date) => Promise<unknown>,
   536→  keep: number,
   537→): Promise<void> {
   538→  try {
   539→    const rows = await findRecent(1, keep - 1)
   540→    if (rows.length === 0) return
   541→    await deleteOlderThan(rows[0].createdAt)
   542→  } catch (error) {
   543→    console.error('[engine/core] prune failed:', error)
   544→  }
   545→}

   547→async function pruneAll(): Promise<void> {
   548→  await pruneTable(
   549→    (take, skip) => db.rawSignal.findMany({ orderBy: { createdAt: 'desc' }, take, skip, select: { createdAt: true } }),
   550→    (createdAt) => db.rawSignal.deleteMany({ where: { createdAt: { lt: createdAt } } }),
   551→    PRUNE_TARGETS.rawSignal,
   552→  )
   553→  await pruneTable(
   554→    (take, skip) => db.crossDomainLink.findMany({ orderBy: { createdAt: 'desc' }, take, skip, select: { createdAt: true } }),
   555→    (createdAt) => db.crossDomainLink.deleteMany({ where: { createdAt: { lt: createdAt } } }),
   556→    PRUNE_TARGETS.crossDomainLink,
   557→  )
   558→  await pruneTable(
   559→    (take, skip) => db.layerMetric.findMany({ orderBy: { createdAt: 'desc' }, take, skip, select: { createdAt: true } }),
   560→    (createdAt) => db.layerMetric.deleteMany({ where: { createdAt: { lt: createdAt } } }),
   561→    PRUNE_TARGETS.layerMetric,
   562→  )
   563→  await pruneTable(
   564→    (take, skip) => db.insightReport.findMany({ orderBy: { createdAt: 'desc' }, take, skip, select: { createdAt: true } }),
   565→    (createdAt) => db.insightReport.deleteMany({ where: { createdAt: { lt: createdAt } } }),
   566→    PRUNE_TARGETS.insightReport,
   567→  )
   568→  await pruneTable(
   569→    (take, skip) => db.governanceCheck.findMany({ orderBy: { createdAt: 'desc' }, take, skip, select: { createdAt: true } }),
   570→    (createdAt) => db.governanceCheck.deleteMany({ where: { createdAt: { lt: createdAt } } }),
   571→    PRUNE_TARGETS.governanceCheck,
   572→  )
   573→}

   575→// ─── runTick: the full 8-layer pipeline ──────────────────────────────────────
   576→
   577→async function runTick(): Promise<TickResult> {
   578→  const startedAt = Date.now()
   579→  const state = await ensureEngine()
   580→  const config = parseConfig(state.config)
   581→  const cycle = state.cycles + 1
   582→
   583→  // ── Layer 1 + 2: INPUT UNIVERSE + PREPROCESSING ──
   584→  const ingest = await runInputLayers(cycle)
   585→  const metricWrites: Array<{ layer: number; key: string; value: number }> = [
   586→    { layer: 1, key: 'ingested', value: ingest.ingested },
   587→    { layer: 1, key: 'streams', value: ingest.streams },
   588→    { layer: 1, key: 'anomalies', value: ingest.anomalies },
   589→    { layer: 2, key: 'promoted', value: ingest.promoted },
   590→    { layer: 2, key: 'avgAnomaly', value: ingest.avgAnomaly },
   591→    { layer: 2, key: 'avgNovelty', value: ingest.avgNovelty },
   592→  ]
   593→
   594→  // ── Layer 3: MULTI-DOMAIN ANALYSIS ──
   595→  const streamRows = await db.domainStream.findMany({ where: { status: 'active' }, select: { category: true } })
   596→  const domainPool = streamRows.map((s) => s.category).filter(isCategory)
   597→  const linkCount = randInt(2, 4)
   598→  for (let i = 0; i < linkCount; i++) {
   599→    let source = domainPool.length >= 2 ? pick(domainPool) : pick(CATEGORIES)
   600→    let target = domainPool.length >= 2 ? pick(domainPool) : pick(CATEGORIES)
   601→    let guard = 0
   602→    while (target === source && guard++ < 10) target = domainPool.length >= 2 ? pick(domainPool) : pick(CATEGORIES)
   603→    if (target === source) continue
   604→    await db.crossDomainLink.create({
   605→      data: {
   606→        sourceDomain: source,
   607→        targetDomain: target,
   608→        strength: round3(rand(0.4, 0.98)),
   609→        mechanism: mechanismBetween(source, target),
   610→        cycle,
   611→      },
   612→    })
   613→  }
   614→  const velocity = round3(clamp01(rand(0.15, 0.85) + ingest.avgAnomaly * 0.2))
   615→  const acceleration = round3(rand(-0.2, 0.35))
   616→  metricWrites.push(
   617→    { layer: 3, key: 'crossLinks', value: linkCount },
   618→    { layer: 3, key: 'velocity', value: velocity },
   619→    { layer: 3, key: 'acceleration', value: acceleration },
   620→  )
   621→
   622→  // ── Layer 4: DISCOVERY (LLM-enhanced, template fallback) ──
   623→  const { hypotheses, llmUsed } = await runDiscoveryLayer(cycle, config.aggression, config.llmDiscovery)
   624→  const validated = hypotheses.filter((h) => h.status === 'validated')
   625→  metricWrites.push(
   626→    { layer: 4, key: 'generated', value: hypotheses.length },
   627→    { layer: 4, key: 'validated', value: validated.length },
   628→  )
   629→
   630→  // ── Layer 5: BREAKTHROUGH SYNTHESIS ──
   631→  const newBreakthroughs: Array<{
   632→    id: string
   633→    title: string
   634→    solutionType: string
   635→    impactRank: number
   636→    gov: ReturnType<typeof buildGovernanceContent>
   637→  }> = []
   638→  for (const h of validated) {
   639→    if (h.data.impact < 0.6) continue
   640→    const solutionType = deriveSolutionType(h.data.domains)
   641→    const synthesis = buildSynthesis(solutionType, h.data.domains)
   642→    const impactRank = round3(h.data.impact * 0.7 + h.data.novelty * 0.3)
   643→    const gov = buildGovernanceContent(solutionType, h.data.domains)
   644→    const breakthrough = await db.breakthrough.create({
   645→      data: {
   646→        hypothesisId: h.id,
   647→        title: h.data.title.slice(0, 300),
   648→        solutionType,
   649→        summary: buildBreakthroughSummary(solutionType, synthesis),
   650→        synthesis: JSON.stringify(synthesis),
   651→        impactRank,
   652→        riskScore: gov.riskScore,
   653→        status: 'synthesized',
   654→        cycle,
   655→      },
   656→    })
   657→    newBreakthroughs.push({
   658→      id: breakthrough.id,
   659→      title: breakthrough.title,
   660→      solutionType,
   661→      impactRank,
   662→      gov,
   663→    })
   664→    await db.hypothesis.update({ where: { id: h.id }, data: { status: 'promoted' } })
   665→  }
   666→  metricWrites.push({ layer: 5, key: 'synthesized', value: newBreakthroughs.length })
   667→
   668→  // ── Layer 6: AUTONOMOUS OPTIMIZATION ──
   669→  const successRate = hypotheses.length > 0 ? round3(validated.length / hypotheses.length) : 0
   670→  const blindSpotScore = round3(clamp01(rand(0.05, 0.6)))
   671→  metricWrites.push(
   672→    { layer: 6, key: 'successRate', value: successRate },
   673→    { layer: 6, key: 'avgNovelty', value: ingest.avgNovelty },
   674→    { layer: 6, key: 'blindSpotScore', value: blindSpotScore },
   675→  )
   676→
   677→  // ── Layer 7: GOVERNANCE ──
   678→  let governed = 0
   679→  let flagged = 0
   680→  for (const bt of newBreakthroughs) {
   681→    const passed = bt.gov.riskScore < 0.35
   682→    await db.governanceCheck.create({
   683→      data: {
   684→        targetType: 'breakthrough',
   685→        targetId: bt.id,
   686→        targetTitle: bt.title,
   687→        riskScore: bt.gov.riskScore,
   688→        ethicalFlags: JSON.stringify(bt.gov.ethicalFlags),
   689→        validationStatus: passed ? 'passed' : 'flagged',
   690→        humanReview: !passed,
   691→        notes: bt.gov.notes,
   692→        cycle,
   693→      },
   694→    })
   695→    if (passed) {
   696→      governed++
   697→      await db.breakthrough.update({ where: { id: bt.id }, data: { status: 'governed' } })
   698→    } else {
   699→      flagged++
   700→    }
   701→  }
   702→  metricWrites.push(
   703→    { layer: 7, key: 'checked', value: newBreakthroughs.length },
   704→    { layer: 7, key: 'passed', value: governed },
   705→    { layer: 7, key: 'flagged', value: flagged },
   706→  )
   707→
   708→  // ── Layer 8: INSIGHT OUTPUT ──
   709→  // cycle-summary every cycle + alerts (impactRank > 0.8) + blind-spot deep-dive (10%)
   710→  const reports: Array<{ title: string; reportType: string; content: string }> = [
   711→    {
   712→      title: `Cycle ${cycle} Summary`,
   713→      reportType: 'cycle-summary',
   714→      content: buildCycleSummary({
   715→        cycle,
   716→        signals: ingest.ingested,
   717→        anomalies: ingest.anomalies,
   718→        hypotheses: hypotheses.length,
   719→        validated: validated.length,
   720→        breakthroughs: newBreakthroughs.length,
   721→        governed,
   722→        flagged,
   723→      }),
   724→    },
   725→  ]
   726→  for (const bt of newBreakthroughs) {
   727→    if (bt.impactRank > 0.8) {
   728→      const alert = buildAlertReport({
   729→        title: bt.title,
   730→        impactRank: bt.impactRank,
   731→        solutionType: bt.solutionType,
   732→        riskScore: bt.gov.riskScore,
   733→        status: bt.gov.riskScore < 0.35 ? 'passed' : 'flagged',
   734→      })
   735→      reports.push({ ...alert, reportType: 'alert' })
   736→    }
   737→  }
   738→  if (chance(0.1)) {
   739→    const weakest = await db.domainStream.findFirst({ orderBy: { health: 'asc' } })
   740→    if (weakest) {
   741→      const deepDive = buildBlindSpotReport({
   742→        cycle,
   743→        streamName: weakest.name,
   744→        health: weakest.health,
   745→        velocity: weakest.velocity,
   746→        successRate,
   747→      })
   748→      reports.push({ ...deepDive, reportType: 'deep-dive' })
   749→    }
   750→  }
   751→
   752→  const persistedReports = await Promise.all(
   753→    reports.map((r) =>
   754→      db.insightReport.create({
   755→        data: { title: r.title, reportType: r.reportType, content: r.content, cycle },
   756→      }),
   757→    ),
   758→  )
   759→  const alertCount = persistedReports.filter((r) => r.reportType === 'alert').length
   760→  metricWrites.push(
   761→    { layer: 8, key: 'reportsPublished', value: persistedReports.length },
   762→    { layer: 8, key: 'alerts', value: alertCount },
   763→  )
   764→
   765→  // Persist all layer metrics.
   766→  await db.layerMetric.createMany({ data: metricWrites.map((m) => ({ ...m, cycle })) })
   767→
   768→  // Close the cycle.
   769→  await db.engineState.update({
   770→    where: { id: ENGINE_ID },
   771→    data: { cycles: cycle, lastTickAt: new Date() },
   772→  })
   773→
   774→  // Prune (best-effort) to keep the SQLite file small.
   775→  await pruneAll()
   776→
   777→  const result: TickResult = {
   778→    cycle,
   779→    signals: ingest.ingested,
   780→    anomalies: ingest.anomalies,
   781→    promoted: ingest.promoted,
   782→    crossLinks: linkCount,
   783→    hypotheses: hypotheses.length,
   784→    validated: validated.length,
   785→    rejected: hypotheses.length - validated.length,
   786→    breakthroughs: newBreakthroughs.length,
   787→    governed,
   788→    flagged,
   789→    reports: persistedReports.length,
   790→    llmUsed,
   791→    durationMs: Date.now() - startedAt,
   792→  }
   793→  console.log(
   794→    `[engine/core] cycle ${cycle} done in ${result.durationMs}ms — signals=${result.signals} ` +
   795→      `hypotheses=${result.hypotheses} (validated=${result.validated}) breakthroughs=${result.breakthroughs} ` +
   796→      `llm=${result.llmUsed ? 'on' : 'off'}`,
   797→  )
   798→  return result
   799→}

   803→interface LayerMetricBundle {
   804→  metrics: Record<string, number>
   805→  hasActivity: boolean
   806→}
   807→
   808→function buildLayers(
   809→  engineStatus: string,
   810→  latestCycle: number,
   811→  byLayer: Map<number, Record<string, number>>,
   812→): LayerStatus[] {
   813→  return LAYER_DEFS.map((def) => {
   814→    const metrics: Record<string, number> = {}
   815→    for (const key of LAYER_METRIC_KEYS[def.key]) {
   816→      metrics[key] = round2(byLayer.get(def.layer)?.[key] ?? 0)
   817→    }
   818→    const hasActivity = byLayer.has(def.layer) && latestCycle > 0
   819→    const loadCfg = LAYER_LOAD[def.key]
   820→    const load = clamp01((byLayer.get(def.layer)?.[loadCfg.key] ?? 0) / loadCfg.max)
   821→    const state: LayerStatus['state'] =
   822→      engineStatus !== 'running' ? 'blocked' : hasActivity ? 'active' : 'idle'
   823→    return {
   824→      layer: def.layer,
   825→      key: def.key,
   826→      name: def.name,
   827→      purpose: def.purpose,
   828→      state,
   829→      load: round2(load),
   830→      metrics,
   831→    }
   832→  })
   833→}

   835→async function buildAnalytics(): Promise<AnalyticsDTO> {
   836→  const [perCycleAll, perCycleAnom, byCategory] = await Promise.all([
   837→    db.rawSignal.groupBy({ by: ['cycle'], _count: { _all: true } }),
   838→    db.rawSignal.groupBy({ by: ['cycle'], _count: { _all: true }, where: { anomalyScore: { gte: ANOMALY_THRESHOLD } } }),
   839→    db.rawSignal.groupBy({ by: ['category'], _count: { _all: true } }),
   840→  ])
   841→
   842→  const anomalyByCycle = new Map<number, number>()
   843→  for (const row of perCycleAnom) anomalyByCycle.set(row.cycle, row._count._all)
   844→  const signalsPerCycle = perCycleAll
   845→    .map((row) => ({
   846→      cycle: row.cycle,
   847→      signals: row._count._all,
   848→      anomalies: anomalyByCycle.get(row.cycle) ?? 0,
   849→    }))
   850→    .sort((a, b) => a.cycle - b.cycle)
   851→    .slice(-30)
   852→
   853→  const domainDistribution = byCategory
   854→    .map((row) => ({ domain: row.category, count: row._count._all }))
   855→    .sort((a, b) => b.count - a.count)
   856→
   857→  const hypForScatter = await db.hypothesis.findMany({
   858→    orderBy: { createdAt: 'desc' },
   859→    take: 60,
   860→    select: { id: true, title: true, novelty: true, impact: true, status: true },
   861→  })
   862→
   863→  const latestPerLayer = await Promise.all(
   864→    [1, 2, 3, 4, 5, 6, 7, 8].map((layer) =>
   865→      db.layerMetric.findFirst({
   866→        where: { layer },
   867→        orderBy: [{ cycle: 'desc' }, { createdAt: 'desc' }],
   868→        select: { value: true },
   869→      }),
   870→    ),
   871→  )
   872→  const layerThroughput = [1, 2, 3, 4, 5, 6, 7, 8].map((layer, i) => ({
   873→    layer,
   874→    value: round2(latestPerLayer[i]?.value ?? 0),
   875→  }))
   876→
   877→  const optimizationRows = await db.layerMetric.findMany({
   878→    where: { layer: 6 },
   879→    orderBy: { cycle: 'asc' },
   880→    select: { cycle: true, key: true, value: true },
   881→  })
   882→  const optimizationByCycle = new Map<number, { successRate: number; avgNovelty: number }>()
   883→  for (const row of optimizationRows) {
   884→    const entry = optimizationByCycle.get(row.cycle) ?? { successRate: 0, avgNovelty: 0 }
   885→    if (row.key === 'successRate') entry.successRate = round2(row.value)
   886→    if (row.key === 'avgNovelty') entry.avgNovelty = round2(row.value)
   887→    optimizationByCycle.set(row.cycle, entry)
   888→  }
   889→  const optimizationTrend = [...optimizationByCycle.entries()]
   890→    .map(([cycle, v]) => ({ cycle, ...v }))
   891→    .sort((a, b) => a.cycle - b.cycle)
   892→    .slice(-30)
   893→
   894→  return {
   895→    signalsPerCycle,
   896→    domainDistribution,
   897→    noveltyImpact: hypForScatter.map((h) => ({
   898→      id: h.id,
   899→      title: h.title,
   900→      novelty: round2(h.novelty),
   901→      impact: round2(h.impact),
   902→      status: h.status,
   903→    })),
   904→    layerThroughput,
   905→    optimizationTrend,
   906→  }
   907→}

   909→export async function getState(): Promise<EngineStateResponse> {
   910→  // ensureEngine + auto-tick when due (the dashboard polls this every ~4s).
   911→  // The tick is fired in the BACKGROUND (not awaited): LLM discovery cycles can
   912→  // take ~10s, and blocking the response on them would make the UI sluggish.
   913→  // The dashboard picks up the results on its next poll a few seconds later.
   914→  const initial = await ensureEngine()
   915→  const cfg = parseConfig(initial.config)
   916→  const due =
   917→    initial.status === 'running' && Date.now() - initial.lastTickAt.getTime() >= cfg.tickIntervalMs
   918→  if (due && !tickMutex.promise) {
   919→    void runTickSerialized().catch((err) => console.error('[engine] background tick failed:', err))
   920→  }
   921→
   922→  const state = await ensureEngine()
   923→  const config = parseConfig(state.config)
   924→  const engineStatus: 'running' | 'paused' = state.status === 'paused' ? 'paused' : 'running'
   925→

   932→      db.hypothesis.findMany({ orderBy: { createdAt: 'desc' }, take: 24 }),

   935→      db.crossDomainLink.findMany({ orderBy: { createdAt: 'desc' }, take: 20 }),

   963→  const hypotheses: HypothesisDTO[] = hypothesisRows.map((h) => ({
   964→    id: h.id,
   965→    title: h.title,
   966→    summary: h.summary,
   967→    domains: parseStringArray(h.domains),
   968→    novelty: round3(h.novelty),
   969→    impact: round3(h.impact),
   970→    confidence: round3(h.confidence),
   971→    status: h.status,
   972→    counterfactuals: parseCounterfactuals(h.counterfactuals),
   973→    validation: parseValidation(h.validation),
   974→    sourceSignals: parseStringArray(h.sourceSignals),
   975→    generator: h.generator,
   976→    cycle: h.cycle,
   977→    createdAt: h.createdAt.toISOString(),
   978→  }))

   982→    hypothesisId: b.hypothesisId,

   984→    solutionType: b.solutionType,
   985→    summary: b.summary,
   986→    synthesis: parseSynthesis(b.synthesis),
   987→    impactRank: round3(b.impactRank),
   988→    riskScore: round3(b.riskScore),
   989→    status: b.status,
   990→    cycle: b.cycle,
   991→    createdAt: b.createdAt.toISOString(),

   995→    id: g.id,
   996→    targetType: g.targetType,
   997→    targetId: g.targetId,
   998→    targetTitle: g.targetTitle,
   999→    riskScore: round3(g.riskScore),
  1000→    ethicalFlags: parseStringArray(g.ethicalFlags),
  1001→    validationStatus: g.validationStatus,

  1003→    reviewDecision: g.reviewDecision,
  1004→    notes: g.notes,
  1005→    cycle: g.cycle,
  1006→    createdAt: g.createdAt.toISOString(),
  1007→  }))

  1010→    id: l.id,
  1011→    sourceDomain: l.sourceDomain,
  1012→    targetDomain: l.targetDomain,
  1013→    strength: round3(l.strength),

  1018→    id: r.id,
  1019→    title: r.title,
  1020→    reportType: r.reportType,

  1028→  const latestCycle = latestMetric?.cycle ?? 0
  1029→  const latestMetrics = latestCycle > 0 ? await db.layerMetric.findMany({ where: { cycle: latestCycle } }) : []
  1030→  const byLayer = new Map<number, Record<string, number>>()
  1031→  for (const m of latestMetrics) {
  1032→    const bucket = byLayer.get(m.layer) ?? {}
  1033→    bucket[m.key] = m.value
  1034→    byLayer.set(m.layer, bucket)
  1035→  }
  1036→  const layers = buildLayers(engineStatus, latestCycle, byLayer)
  1037→
  1038→  const analytics = await buildAnalytics()
  1039→
  1040→  return {
  1041→    engine: {
  1042→      status: engineStatus,
  1043→      cycles: state.cycles,
  1044→      startedAt: state.startedAt.toISOString(),
  1045→      lastTickAt: state.lastTickAt.toISOString(),
  1046→      uptimeSeconds: Math.max(0, Math.round((Date.now() - state.startedAt.getTime()) / 1000)),
  1047→      config,

  1050→    streams,
  1051→    signals,
  1052→    hypotheses,
  1053→    breakthroughs,
  1054→    governance,
  1055→    correlations,
  1056→    reports,
  1057→    analytics,
  1058→  }
